import cors from '@fastify/cors';
import multipart from '@fastify/multipart';
import fastifyStatic from '@fastify/static';
import { mkdirSync } from 'node:fs';
import path from 'node:path';
import Fastify from 'fastify';
import { sql } from 'drizzle-orm';
import { ZodError } from 'zod';
import type { Db } from './db/client.js';
import { ErroNegocio } from './shared/erros.js';
import { fotosRoutes } from './modules/fotos/fotos.routes.js';
import { familiasRoutes } from './modules/familias/familias.routes.js';
import { pessoasRoutes } from './modules/pessoas/pessoas.routes.js';

export async function createApp(opts: { db: Db; webOrigin: string; uploadDir: string }) {
  const app = Fastify({ logger: true });
  await app.register(cors, {
    origin: opts.webOrigin,
    methods: ['GET', 'HEAD', 'POST', 'PUT', 'DELETE'],
  });

  const uploadDir = path.resolve(opts.uploadDir);
  mkdirSync(path.join(uploadDir, 'fotos'), { recursive: true });
  await app.register(multipart, { limits: { fileSize: 5 * 1024 * 1024, files: 1 } });
  await app.register(fastifyStatic, { root: uploadDir, prefix: '/api/uploads/' });

  app.setErrorHandler((erro, _req, reply) => {
    if (erro instanceof ZodError) {
      return reply.status(400).send({ erro: 'Dados inválidos', detalhes: erro.flatten() });
    }
    if (erro instanceof ErroNegocio) {
      return reply.status(erro.status).send({ erro: erro.message });
    }
    app.log.error(erro);
    return reply.status(500).send({ erro: 'Erro interno' });
  });

  app.get('/api/health', async () => ({ ok: true }));
  app.get('/api/health/db', async () => {
    await opts.db.execute(sql`select 1`);
    return { ok: true, db: 'conectado' };
  });

  await app.register(pessoasRoutes(opts.db), { prefix: '/api' });
  await app.register(familiasRoutes(opts.db), { prefix: '/api' });
  await app.register(fotosRoutes(opts.db, uploadDir), { prefix: '/api' });
  return app;
}

  integer,

  varchar,
} from 'drizzle-orm/pg-core';

export const papelFiliacao = pgEnum('papel_filiacao', ['PAI', 'MAE']);

export const pessoa = pgTable('pessoa', {
  id: integer().primaryKey().generatedAlwaysAsIdentity(),
  nome: varchar({ length: 150 }).notNull(),
  dataNascimento: date('data_nascimento'),
  sexo: varchar({ length: 1 }),
  foto: varchar({ length: 255 }),
  criadoEm: timestamp('criado_em').notNull().defaultNow(),
});

export const uniao = pgTable(
  'uniao',
  {
    id: integer().primaryKey().generatedAlwaysAsIdentity(),
    pessoaAId: integer('pessoa_a_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    pessoaBId: integer('pessoa_b_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
  },
  (t) => [
    check('uniao_ordem', sql`${t.pessoaAId} < ${t.pessoaBId}`),
    unique('uniao_par').on(t.pessoaAId, t.pessoaBId),
  ],
);

export const filiacao = pgTable(
  'filiacao',
  {
    id: integer().primaryKey().generatedAlwaysAsIdentity(),
    filhoId: integer('filho_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    genitorId: integer('genitor_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    papel: papelFiliacao().notNull(),
  },
  (t) => [
    unique('filiacao_filho_papel').on(t.filhoId, t.papel),
    check('filiacao_distintos', sql`${t.filhoId} <> ${t.genitorId}`),
  ],

import { and, eq } from 'drizzle-orm';
import type { Db } from '../../db/client.js';
import { filiacao, uniao } from '../../db/schema.js';

export function familiasRepository(db: Db) {
  return {
    listarUnioes: () => db.select().from(uniao),
    async buscarUniao(a: number, b: number) {
      const [u] = await db
        .select()
        .from(uniao)
        .where(and(eq(uniao.pessoaAId, a), eq(uniao.pessoaBId, b)));
      return u;
    },
    async criarUniao(a: number, b: number) {
      const [u] = await db.insert(uniao).values({ pessoaAId: a, pessoaBId: b }).returning();
      return u!;
    },
    async excluirUniao(id: number) {
      const r = await db.delete(uniao).where(eq(uniao.id, id)).returning({ id: uniao.id });
      return r.length > 0;
    },

    listarFiliacoes: () => db.select().from(filiacao),
    async buscarFiliacaoDoFilho(filhoId: number, papel: 'PAI' | 'MAE') {
      const [f] = await db
        .select()
        .from(filiacao)
        .where(and(eq(filiacao.filhoId, filhoId), eq(filiacao.papel, papel)));
      return f;
    },
    async criarFiliacao(filhoId: number, genitorId: number, papel: 'PAI' | 'MAE') {
      const [f] = await db.insert(filiacao).values({ filhoId, genitorId, papel }).returning();
      return f!;
    },
    async excluirFiliacao(id: number) {
      const r = await db.delete(filiacao).where(eq(filiacao.id, id)).returning({ id: filiacao.id });
      return r.length > 0;
    },
  };
}
export type FamiliasRepository = ReturnType<typeof familiasRepository>;

import type { FastifyInstance } from 'fastify';
import type { Db } from '../../db/client.js';
import { pessoasRepository } from '../pessoas/pessoas.repository.js';
import { criarFiliacaoSchema, criarUniaoSchema, idParamSchema } from './familias.schema.js';
import { familiasRepository } from './familias.repository.js';
import { familiasService } from './familias.service.js';

export function familiasRoutes(db: Db) {
  const service = familiasService(familiasRepository(db), pessoasRepository(db));
  return async (app: FastifyInstance) => {
    app.get('/unioes', async () => service.listarUnioes());
    app.post('/unioes', async (req, reply) =>
      reply.status(201).send(await service.criarUniao(criarUniaoSchema.parse(req.body))),
    );
    app.delete('/unioes/:id', async (req, reply) => {
      await service.excluirUniao(idParamSchema.parse(req.params).id);
      return reply.status(204).send();
    });

    app.get('/filiacoes', async () => service.listarFiliacoes());
    app.post('/filiacoes', async (req, reply) =>
      reply.status(201).send(await service.criarFiliacao(criarFiliacaoSchema.parse(req.body))),
    );
    app.delete('/filiacoes/:id', async (req, reply) => {
      await service.excluirFiliacao(idParamSchema.parse(req.params).id);
      return reply.status(204).send();
    });
  };
}

import { ehAncestral } from '@raizes/domain';
import { ErroNegocio, naoEncontrado } from '../../shared/erros.js';
import type { PessoasRepository } from '../pessoas/pessoas.repository.js';
import type { FamiliasRepository } from './familias.repository.js';
import type { CriarFiliacao, CriarUniao } from './familias.schema.js';

export function familiasService(repo: FamiliasRepository, pessoas: PessoasRepository) {
  const exigirPessoa = async (id: number) => {
    if (!(await pessoas.buscar(id))) throw naoEncontrado(`Pessoa ${id}`);
  };

  return {
    listarUnioes: () => repo.listarUnioes(),
    listarFiliacoes: () => repo.listarFiliacoes(),

    async criarUniao({ pessoaAId, pessoaBId }: CriarUniao) {
      if (pessoaAId === pessoaBId) throw new ErroNegocio('Uma pessoa não pode se unir a ela mesma');
      await exigirPessoa(pessoaAId);
      await exigirPessoa(pessoaBId);
      // sempre grava com a < b, para o casal existir uma vez só
      const a = Math.min(pessoaAId, pessoaBId);
      const b = Math.max(pessoaAId, pessoaBId);
      if (await repo.buscarUniao(a, b)) throw new ErroNegocio('Esse casal já está cadastrado', 409);
      return repo.criarUniao(a, b);
    },
    async excluirUniao(id: number) {
      if (!(await repo.excluirUniao(id))) throw naoEncontrado('União');
    },

    async criarFiliacao({ filhoId, genitorId, papel }: CriarFiliacao) {
      if (filhoId === genitorId) throw new ErroNegocio('Ninguém pode ser pai/mãe de si mesmo');
      await exigirPessoa(filhoId);
      await exigirPessoa(genitorId);
      if (await repo.buscarFiliacaoDoFilho(filhoId, papel)) {
        throw new ErroNegocio(`Essa pessoa já tem ${papel === 'PAI' ? 'um pai' : 'uma mãe'} cadastrado(a)`, 409);
      }
      const todas = await repo.listarFiliacoes();
      // o "filho" não pode ser ancestral do genitor (evita ciclos na árvore)
      if (ehAncestral(todas, filhoId, genitorId)) {
        throw new ErroNegocio('Isso criaria um ciclo: o filho já é ancestral dessa pessoa');
      }
      return repo.criarFiliacao(filhoId, genitorId, papel);
    },
    async excluirFiliacao(id: number) {
      if (!(await repo.excluirFiliacao(id))) throw naoEncontrado('Filiação');
    },
  };
}

import { createWriteStream } from 'node:fs';
import { mkdir, unlink } from 'node:fs/promises';
import path from 'node:path';
import { pipeline } from 'node:stream/promises';
import type { FastifyInstance } from 'fastify';
import type { Db } from '../../db/client.js';
import { ErroNegocio, naoEncontrado } from '../../shared/erros.js';
import { pessoasRepository } from '../pessoas/pessoas.repository.js';
import { idParamSchema } from '../pessoas/pessoas.schema.js';

const EXTENSOES: Record<string, string> = {
  'image/jpeg': '.jpg',
  'image/png': '.png',
  'image/webp': '.webp',
};

export function fotosRoutes(db: Db, uploadDir: string) {
  const pessoas = pessoasRepository(db);
  const dirFotos = path.join(uploadDir, 'fotos');

  const apagarArquivo = (relativo: string | null) =>
    relativo ? unlink(path.join(uploadDir, relativo)).catch(() => undefined) : Promise.resolve();

  return async (app: FastifyInstance) => {
    app.post('/pessoas/:id/foto', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      const pessoa = await pessoas.buscar(id);
      if (!pessoa) throw naoEncontrado('Pessoa');

      const arquivo = await req.file();
      if (!arquivo) throw new ErroNegocio('Envie uma imagem');
      const ext = EXTENSOES[arquivo.mimetype];
      if (!ext) {
        arquivo.file.resume();
        throw new ErroNegocio('Formato inválido: use JPG, PNG ou WebP');
      }

      await mkdir(dirFotos, { recursive: true });
      const nome = `${id}-${Date.now()}${ext}`;
      const destino = path.join(dirFotos, nome);
      await pipeline(arquivo.file, createWriteStream(destino));
      if (arquivo.file.truncated) {
        await unlink(destino).catch(() => undefined);
        throw new ErroNegocio('Imagem maior que 5 MB');
      }

      await apagarArquivo(pessoa.foto);
      return pessoas.definirFoto(id, `fotos/${nome}`);
    });

    app.delete('/pessoas/:id/foto', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      const pessoa = await pessoas.buscar(id);
      if (!pessoa) throw naoEncontrado('Pessoa');
      await apagarArquivo(pessoa.foto);
      return pessoas.definirFoto(id, null);
    });
  };
}

import { asc, eq } from 'drizzle-orm';
import type { Db } from '../../db/client.js';
import { pessoa } from '../../db/schema.js';
import type { CriarPessoa } from './pessoas.schema.js';

const valores = (d: CriarPessoa) => ({
  nome: d.nome,
  dataNascimento: d.dataNascimento ?? null,
  sexo: d.sexo ?? null,
});

export function pessoasRepository(db: Db) {
  return {
    listar: () => db.select().from(pessoa).orderBy(asc(pessoa.nome)),
    async buscar(id: number) {
      const [p] = await db.select().from(pessoa).where(eq(pessoa.id, id));
      return p;
    },
    async criar(dados: CriarPessoa) {
      const [p] = await db.insert(pessoa).values(valores(dados)).returning();
      return p!;
    },
    async atualizar(id: number, dados: CriarPessoa) {
      const [p] = await db.update(pessoa).set(valores(dados)).where(eq(pessoa.id, id)).returning();
      return p;
    },
    async definirFoto(id: number, caminho: string | null) {
      const [p] = await db.update(pessoa).set({ foto: caminho }).where(eq(pessoa.id, id)).returning();
      return p;
    },
    async excluir(id: number) {
      const r = await db.delete(pessoa).where(eq(pessoa.id, id)).returning({ id: pessoa.id });
      return r.length > 0;
    },
  };
}
export type PessoasRepository = ReturnType<typeof pessoasRepository>;

import type { FastifyInstance } from 'fastify';
import type { Db } from '../../db/client.js';
import { atualizarPessoaSchema, criarPessoaSchema, idParamSchema } from './pessoas.schema.js';
import { pessoasRepository } from './pessoas.repository.js';
import { pessoasService } from './pessoas.service.js';

export function pessoasRoutes(db: Db) {
  const service = pessoasService(pessoasRepository(db));
  return async (app: FastifyInstance) => {
    app.get('/pessoas', async () => service.listar());
    app.get('/pessoas/:id', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      return service.obter(id);
    });
    app.post('/pessoas', async (req, reply) => {
      const dados = criarPessoaSchema.parse(req.body);
      return reply.status(201).send(await service.criar(dados));
    });
    app.put('/pessoas/:id', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      return service.atualizar(id, atualizarPessoaSchema.parse(req.body));
    });
    app.delete('/pessoas/:id', async (req, reply) => {
      const { id } = idParamSchema.parse(req.params);
      await service.excluir(id);
      return reply.status(204).send();
    });
  };
}

export async function api<T>(caminho: string, init?: RequestInit): Promise<T> {
  const resp = await fetch(`/api${caminho}`, {
    ...init,
    headers: init?.body ? { 'Content-Type': 'application/json' } : undefined,
  });
  if (!resp.ok) {
    const corpo = (await resp.json().catch(() => null)) as { erro?: string } | null;
    throw new Error(corpo?.erro ?? `Erro ${resp.status}`);
  }
  if (resp.status === 204) return undefined as T;
  if (resp.status === 204) return undefined as T;
  return resp.json() as Promise<T>;
}

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { useState, type FormEvent } from 'react';
import { api } from '../api/cliente';

type PessoaDto = { id: number; nome: string };
type UniaoDto = { id: number; pessoaAId: number; pessoaBId: number };
type FiliacaoDto = { id: number; filhoId: number; genitorId: number; papel: 'PAI' | 'MAE' };

function SelectPessoa(props: {
  valor: string;
  onChange: (v: string) => void;
  pessoas: PessoaDto[];
  rotulo: string;
}) {
  return (
    <select value={props.valor} onChange={(e) => props.onChange(e.target.value)} required>
      <option value="">{props.rotulo}</option>
      {props.pessoas.map((p) => (
        <option key={p.id} value={p.id}>
          {p.nome}
        </option>
      ))}
    </select>
  );
}

export function Familias() {
  const qc = useQueryClient();
  const pessoas = useQuery({ queryKey: ['pessoas'], queryFn: () => api<PessoaDto[]>('/pessoas') });
  const unioes = useQuery({ queryKey: ['unioes'], queryFn: () => api<UniaoDto[]>('/unioes') });
  const filiacoes = useQuery({
    queryKey: ['filiacoes'],
    queryFn: () => api<FiliacaoDto[]>('/filiacoes'),
  });

  const nomes = new Map((pessoas.data ?? []).map((p) => [p.id, p.nome]));

# Arquitetura

```
raizes-memorias/
├─ packages/domain   TypeScript puro: tipos, nomes curtos, (depois) layout da árvore e parentesco
├─ apps/api          Fastify + Drizzle + PostgreSQL
│   └─ modules/<x>   routes → service → repository
└─ apps/web          React + Vite + TanStack Query
```

Regras:
- `domain` não importa nada de api/web nem de bibliotecas de runtime; testável sozinho.
- API em camadas: **routes** (HTTP + zod) → **service** (regras) → **repository** (SQL via Drizzle).
- Tudo por **id**, nunca por nome. Parentesco é **derivado**, não gravado.
- `createApp({ db })` permite testar a API com banco de teste.
- Web fala com a API só via `src/api/cliente.ts`; dados do servidor via TanStack Query.
- Fotos: caminho relativo em `pessoa.foto`, arquivos em `apps/api/uploads`.

import type { Filiacao } from '../tipos';

/**
 * Verdadeiro se `ancestralId` é ancestral (pai, avô, bisavô...) de `pessoaId`.
 * Usado para impedir ciclos: ninguém pode ser pai de um de seus próprios ancestrais.
 */
export function ehAncestral(
  filiacoes: Pick<Filiacao, 'filhoId' | 'genitorId'>[],
  ancestralId: number,
  pessoaId: number,
): boolean {
  const genitoresDe = new Map<number, number[]>();
  for (const f of filiacoes) {
    const lista = genitoresDe.get(f.filhoId) ?? [];
    lista.push(f.genitorId);
    genitoresDe.set(f.filhoId, lista);
  }
  const visitados = new Set<number>();
  const pilha = [pessoaId];
  while (pilha.length > 0) {
    const atual = pilha.pop()!;
    for (const g of genitoresDe.get(atual) ?? []) {
      if (g === ancestralId) return true;
      if (!visitados.has(g)) {
        visitados.add(g);
        pilha.push(g);
      }
    }
  }
  return false;
}

const CONECTORES = new Set(['de', 'da', 'do', 'das', 'dos', 'e']);

function normalizar(texto: string): string {
  return texto.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase();
}

/** As primeiras n palavras "de verdade" (de/da/do não contam, mas ficam no texto). */
function comPalavras(completo: string, n: number): string {
  const partes = completo.trim().split(/\s+/);
  const usadas: string[] = [];
  let contadas = 0;

  for (const parte of partes) {
    const conector = CONECTORES.has(parte.toLowerCase());
    if (!conector && contadas === n) break;
    usadas.push(parte);
    if (!conector) contadas++;
  }
  while (usadas.length > 0 && CONECTORES.has(usadas[usadas.length - 1]!.toLowerCase())) {
    usadas.pop();
  }

  const texto = usadas.join(' ');
  // nomes muito curtos (ex.: "Zé") ganham a segunda palavra
  if (n === 1 && partes.length > 1 && texto.length <= 2) return comPalavras(completo, 2);
  return texto;
}

/**
 * Cada pessoa mostra o menor nome que a diferencia das outras
 * (Maria, Maria Celina, Maria do Socorro...). Portado do ArvoreController em Java.
 */
export function nomesCurtos(pessoas: { id: number; nome: string }[]): Map<number, string> {
  const resultado = new Map<number, string>();
  const validas = pessoas.filter((p) => p.nome.trim() !== '');

  for (let palavras = 1; palavras <= 12; palavras++) {
    const candidatos = new Map<number, string>();
    const contagem = new Map<string, number>();

    for (const p of validas) {
      const nome = comPalavras(p.nome, palavras);
      candidatos.set(p.id, nome);
      const chave = normalizar(nome);
      contagem.set(chave, (contagem.get(chave) ?? 0) + 1);
    }
    for (const p of validas) {
      if (resultado.has(p.id)) continue;
      const nome = candidatos.get(p.id)!;
      if (contagem.get(normalizar(nome)) === 1 || palavras === 12) resultado.set(p.id, nome);
    }
  }
  return resultado;
}

/**
 * Tipos do domínio. Este pacote NÃO conhece banco, HTTP nem tela:
 * só regras e cálculos puros, fáceis de testar.

export type Sexo = 'M' | 'F';

/** Papel de um genitor em relação ao filho. Não depende do sexo da pessoa. */
export type PapelFiliacao = 'PAI' | 'MAE';

export interface Pessoa {
  id: number;
  nome: string;
  /** Formato AAAA-MM-DD */
  dataNascimento: string | null;
  sexo: Sexo | null;
  /** Caminho RELATIVO da foto (ex.: "fotos/benedito.png") */
  foto: string | null;
}

/** Casal. Sempre guardado com pessoaAId < pessoaBId, para não duplicar. */
export interface Uniao {
  id: number;
  pessoaAId: number;
  pessoaBId: number;
}

/** "genitorId é PAI/MAE de filhoId". Cada filho tem no máximo um PAI e uma MAE. */
export interface Filiacao {
  id: number;
  filhoId: number;
  genitorId: number;
  papel: PapelFiliacao;
}

/** Uma família montada por id: quem são os pais e quais são os filhos. */
export interface Familia {
  paiId: number | null;
  maeId: number | null;
  filhosIds: number[];
}
