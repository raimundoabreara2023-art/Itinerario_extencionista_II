import cors from '@fastify/cors';
import multipart from '@fastify/multipart';
import fastifyStatic from '@fastify/static';
import { mkdirSync } from 'node:fs';
import path from 'node:path';
import Fastify from 'fastify';
import { sql } from 'drizzle-orm';
import { ZodError } from 'zod';
import type { Db } from './db/client.js';
import { ErroNegocio } from './shared/erros.js';
import { fotosRoutes } from './modules/fotos/fotos.routes.js';
import { familiasRoutes } from './modules/familias/familias.routes.js';
import { pessoasRoutes } from './modules/pessoas/pessoas.routes.js';

export async function createApp(opts: { db: Db; webOrigin: string; uploadDir: string }) {
  const app = Fastify({ logger: true });
  await app.register(cors, {
    origin: opts.webOrigin,
    methods: ['GET', 'HEAD', 'POST', 'PUT', 'DELETE'],
  });

  const uploadDir = path.resolve(opts.uploadDir);
  mkdirSync(path.join(uploadDir, 'fotos'), { recursive: true });
  await app.register(multipart, { limits: { fileSize: 5 * 1024 * 1024, files: 1 } });
  await app.register(fastifyStatic, { root: uploadDir, prefix: '/api/uploads/' });

  app.setErrorHandler((erro, _req, reply) => {
    if (erro instanceof ZodError) {
      return reply.status(400).send({ erro: 'Dados inválidos', detalhes: erro.flatten() });
    }
    if (erro instanceof ErroNegocio) {
      return reply.status(erro.status).send({ erro: erro.message });
    }
    app.log.error(erro);
    return reply.status(500).send({ erro: 'Erro interno' });
  });

  app.get('/api/health', async () => ({ ok: true }));
  app.get('/api/health/db', async () => {
    await opts.db.execute(sql`select 1`);
    return { ok: true, db: 'conectado' };
  });

  await app.register(pessoasRoutes(opts.db), { prefix: '/api' });
  await app.register(familiasRoutes(opts.db), { prefix: '/api' });
  await app.register(fotosRoutes(opts.db, uploadDir), { prefix: '/api' });
  return app;
}

export const papelFiliacao = pgEnum('papel_filiacao', ['PAI', 'MAE']);

export const pessoa = pgTable('pessoa', {
  id: integer().primaryKey().generatedAlwaysAsIdentity(),
  nome: varchar({ length: 150 }).notNull(),
  dataNascimento: date('data_nascimento'),
  sexo: varchar({ length: 1 }),
  foto: varchar({ length: 255 }),
  criadoEm: timestamp('criado_em').notNull().defaultNow(),
});

export const uniao = pgTable(
  'uniao',
  {
    id: integer().primaryKey().generatedAlwaysAsIdentity(),
    pessoaAId: integer('pessoa_a_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    pessoaBId: integer('pessoa_b_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
  },
  (t) => [
    check('uniao_ordem', sql`${t.pessoaAId} < ${t.pessoaBId}`),
    unique('uniao_par').on(t.pessoaAId, t.pessoaBId),
  ],
);

export const filiacao = pgTable(
  'filiacao',
  {
    id: integer().primaryKey().generatedAlwaysAsIdentity(),
    filhoId: integer('filho_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    genitorId: integer('genitor_id').notNull().references(() => pessoa.id, { onDelete: 'cascade' }),
    papel: papelFiliacao().notNull(),
  },
  (t) => [
    unique('filiacao_filho_papel').on(t.filhoId, t.papel),
    check('filiacao_distintos', sql`${t.filhoId} <> ${t.genitorId}`),

import { and, eq } from 'drizzle-orm';
import type { Db } from '../../db/client.js';
import { filiacao, uniao } from '../../db/schema.js';

export function familiasRepository(db: Db) {
  return {
    listarUnioes: () => db.select().from(uniao),
    async buscarUniao(a: number, b: number) {
      const [u] = await db
        .select()
        .from(uniao)
        .where(and(eq(uniao.pessoaAId, a), eq(uniao.pessoaBId, b)));
      return u;
    },
    async criarUniao(a: number, b: number) {
      const [u] = await db.insert(uniao).values({ pessoaAId: a, pessoaBId: b }).returning();
      return u!;
    },
    async excluirUniao(id: number) {
      const r = await db.delete(uniao).where(eq(uniao.id, id)).returning({ id: uniao.id });
      return r.length > 0;
    },

    listarFiliacoes: () => db.select().from(filiacao),
    async buscarFiliacaoDoFilho(filhoId: number, papel: 'PAI' | 'MAE') {
      const [f] = await db
        .select()
        .from(filiacao)
        .where(and(eq(filiacao.filhoId, filhoId), eq(filiacao.papel, papel)));
      return f;
    },
    async criarFiliacao(filhoId: number, genitorId: number, papel: 'PAI' | 'MAE') {
      const [f] = await db.insert(filiacao).values({ filhoId, genitorId, papel }).returning();
      return f!;
    },
    async excluirFiliacao(id: number) {
      const r = await db.delete(filiacao).where(eq(filiacao.id, id)).returning({ id: filiacao.id });
      return r.length > 0;
    },
  };
}
export type FamiliasRepository = ReturnType<typeof familiasRepository>;

import type { FastifyInstance } from 'fastify';
import type { Db } from '../../db/client.js';
import { pessoasRepository } from '../pessoas/pessoas.repository.js';
import { criarFiliacaoLoteSchema, criarUniaoSchema, idParamSchema } from './familias.schema.js';
import { familiasRepository } from './familias.repository.js';
import { familiasService } from './familias.service.js';

export function familiasRoutes(db: Db) {
  const service = familiasService(familiasRepository(db), pessoasRepository(db));
  return async (app: FastifyInstance) => {
    app.get('/unioes', async () => service.listarUnioes());
    app.post('/unioes', async (req, reply) =>
      reply.status(201).send(await service.criarUniao(criarUniaoSchema.parse(req.body))),
    );
    app.delete('/unioes/:id', async (req, reply) => {
      await service.excluirUniao(idParamSchema.parse(req.params).id);
      return reply.status(204).send();
    });

    app.get('/filiacoes', async () => service.listarFiliacoes());
    app.post('/filiacoes', async (req, reply) =>
      reply.status(201).send(await service.criarFiliacaoLote(criarFiliacaoLoteSchema.parse(req.body))),
    );
    app.delete('/filiacoes/:id', async (req, reply) => {
      await service.excluirFiliacao(idParamSchema.parse(req.params).id);
      return reply.status(204).send();

import { ehAncestral } from '@raizes/domain';
import { ErroNegocio, naoEncontrado } from '../../shared/erros.js';
import type { PessoasRepository } from '../pessoas/pessoas.repository.js';
import type { FamiliasRepository } from './familias.repository.js';
import type { CriarFiliacaoLote, CriarUniao } from './familias.schema.js';

export function familiasService(repo: FamiliasRepository, pessoas: PessoasRepository) {
  const exigirPessoa = async (id: number) => {
    if (!(await pessoas.buscar(id))) throw naoEncontrado(`Pessoa ${id}`);
  };

  return {
    listarUnioes: () => repo.listarUnioes(),
    listarFiliacoes: () => repo.listarFiliacoes(),

    async criarUniao({ pessoaAId, pessoaBId }: CriarUniao) {
      if (pessoaAId === pessoaBId) throw new ErroNegocio('Uma pessoa não pode se unir a ela mesma');
      await exigirPessoa(pessoaAId);
      await exigirPessoa(pessoaBId);
      const a = Math.min(pessoaAId, pessoaBId);
      const b = Math.max(pessoaAId, pessoaBId);
      if (await repo.buscarUniao(a, b)) throw new ErroNegocio('Esse casal já está cadastrado', 409);
      return repo.criarUniao(a, b);
    },

    async excluirUniao(id: number) {
      if (!(await repo.excluirUniao(id))) throw naoEncontrado('União');
    },

    async criarFiliacaoLote({ filhosIds, genitorId, papel }: CriarFiliacaoLote) {
      await exigirPessoa(genitorId);
      const todas = await repo.listarFiliacoes();
      const resultados = [];

      for (const filhoId of filhosIds) {
        if (filhoId === genitorId) {
          throw new ErroNegocio('Ninguém pode ser pai/mãe de si mesmo');
        }
        await exigirPessoa(filhoId);

        if (await repo.buscarFiliacaoDoFilho(filhoId, papel)) {
          const nomePapel = papel === 'PAI' ? 'um pai' : 'uma mãe';
          throw new ErroNegocio(`Uma das pessoas selecionadas já tem ${nomePapel} cadastrado(a)`, 409);
        }

        if (ehAncestral(todas, filhoId, genitorId)) {
          throw new ErroNegocio('Isso criaria um ciclo: o filho já é ancestral dessa pessoa');
        }

        const nova = await repo.criarFiliacao(filhoId, genitorId, papel);
        resultados.push(nova);
        todas.push({ id: nova.id, filhoId, genitorId, papel });
      }

      return resultados;
    },

    async excluirFiliacao(id: number) {
      if (!(await repo.excluirFiliacao(id))) throw naoEncontrado('Filiação');
    },
  };
}

import { createWriteStream } from 'node:fs';
import { mkdir, unlink } from 'node:fs/promises';
import path from 'node:path';
import { pipeline } from 'node:stream/promises';
import type { FastifyInstance } from 'fastify';
import type { Db } from '../../db/client.js';
import { ErroNegocio, naoEncontrado } from '../../shared/erros.js';
import { pessoasRepository } from '../pessoas/pessoas.repository.js';
import { idParamSchema } from '../pessoas/pessoas.schema.js';

const EXTENSOES: Record<string, string> = {
  'image/jpeg': '.jpg',
  'image/png': '.png',
  'image/webp': '.webp',
};

export function fotosRoutes(db: Db, uploadDir: string) {
  const pessoas = pessoasRepository(db);
  const dirFotos = path.join(uploadDir, 'fotos');

  const apagarArquivo = (relativo: string | null) =>
    relativo ? unlink(path.join(uploadDir, relativo)).catch(() => undefined) : Promise.resolve();

  return async (app: FastifyInstance) => {
    app.post('/pessoas/:id/foto', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      const pessoa = await pessoas.buscar(id);
      if (!pessoa) throw naoEncontrado('Pessoa');

      const arquivo = await req.file();
      if (!arquivo) throw new ErroNegocio('Envie uma imagem');
      const ext = EXTENSOES[arquivo.mimetype];
      if (!ext) {
        arquivo.file.resume();
        throw new ErroNegocio('Formato inválido: use JPG, PNG ou WebP');
      }

      await mkdir(dirFotos, { recursive: true });
      const nome = `${id}-${Date.now()}${ext}`;
      const destino = path.join(dirFotos, nome);
      await pipeline(arquivo.file, createWriteStream(destino));
      if (arquivo.file.truncated) {
        await unlink(destino).catch(() => undefined);
        throw new ErroNegocio('Imagem maior que 5 MB');
      }

      await apagarArquivo(pessoa.foto);
      return pessoas.definirFoto(id, `fotos/${nome}`);
    });

    app.delete('/pessoas/:id/foto', async (req) => {
      const { id } = idParamSchema.parse(req.params);
      const pessoa = await pessoas.buscar(id);
      if (!pessoa) throw naoEncontrado('Pessoa');
      await apagarArquivo(pessoa.foto);
      return pessoas.definirFoto(id, null);
    });
  };
}

import { asc, eq } from 'drizzle-orm';
import type { Db } from '../../db/client.js';
import { pessoa } from '../../db/schema.js';
import type { CriarPessoa } from './pessoas.schema.js';

const valores = (d: CriarPessoa) => ({
  nome: d.nome,
  dataNascimento: d.dataNascimento ?? null,
  sexo: d.sexo ?? null,
});

export function pessoasRepository(db: Db) {
  return {
    listar: () => db.select().from(pessoa).orderBy(asc(pessoa.nome)),
    async buscar(id: number) {
      const [p] = await db.select().from(pessoa).where(eq(pessoa.id, id));
      return p;
    },
    async criar(dados: CriarPessoa) {
      const [p] = await db.insert(pessoa).values(valores(dados)).returning();
      return p!;
    },
    async atualizar(id: number, dados: CriarPessoa) {
      const [p] = await db.update(pessoa).set(valores(dados)).where(eq(pessoa.id, id)).returning();
      return p;
    },
    async definirFoto(id: number, caminho: string | null) {
      const [p] = await db.update(pessoa).set({ foto: caminho }).where(eq(pessoa.id, id)).returning();
      return p;
    },
    async excluir(id: number) {
      const r = await db.delete(pessoa).where(eq(pessoa.id, id)).returning({ id: pessoa.id });
      return r.length > 0;
    },
  };
}
export type PessoasRepository = ReturnType<typeof pessoasRepository>;

export async function api<T>(caminho: string, init?: RequestInit): Promise<T> {
  const resp = await fetch(`/api${caminho}`, {
    ...init,
    headers: init?.body ? { 'Content-Type': 'application/json' } : undefined,
  });
  if (!resp.ok) {
    const corpo = (await resp.json().catch(() => null)) as { erro?: string } | null;
    throw new Error(corpo?.erro ?? `Erro ${resp.status}`);
  }
  if (resp.status === 204) return undefined as T;
  return resp.json() as Promise<T>;
}

  const dados = useMemo(() => {
    if (!pessoas.data || !unioes.data || !filiacoes.data) return null;
    const familias: Familia[] = montarFamilias(unioes.data, filiacoes.data);
    const layout = calcularLayout(familias);
    const porId = new Map(pessoas.data.map((p) => [p.id, p]));
    const noLayout = pessoas.data.filter((p) => layout.x.has(p.id));
    const curtos = nomesCurtos(noLayout);
    const semLigacao = pessoas.data.filter((p) => !layout.x.has(p.id));
    return { layout, porId, curtos, semLigacao };
  }, [pessoas.data, unioes.data, filiacoes.data]);

  // pessoa em foco: por padrão, a primeira da última geração
  const focoPadrao = useMemo(() => {
    if (!dados) return null;
    let melhor: number | null = null;
    let maior = -1;
    for (const [id, linha] of dados.layout.linha) {
      if (linha > maior) {
        maior = linha;
        melhor = id;
      }
    }
    return melhor;
  }, [dados]);
  const foco = focoEscolhido === undefined ? focoPadrao : focoEscolhido;

  const linhaDireta = useMemo(() => {
    const direta = new Set<number>();
    if (!dados || foco === null) return direta;
    const pilha = [foco];
    while (pilha.length > 0) {
      const atual = pilha.pop()!;
      if (direta.has(atual)) continue;
      direta.add(atual);
      pilha.push(...(dados.layout.pais.get(atual) ?? []));
    }
    return direta;
  }, [dados, foco]);

  const largura = dados ? dados.layout.largura + MARGEM_ROTULOS : 0;
  const altura = dados ? TOPO + dados.layout.totalLinhas * ALTURA_LINHA + 90 : 0;

  const zoomParaLargura = () => {
    const w = areaRef.current?.clientWidth ?? 0;
    if (w <= 0 || largura <= 0) return 1;
    return Math.min(1, (w - 30) / largura);
  };
  const zoomParaTudo = () => {
    const h = areaRef.current?.clientHeight ?? 0;
    if (h <= 0 || altura <= 0) return zoomParaLargura();
    return Math.min(zoomParaLargura(), (h - 30) / altura);
  };
  const limitar = (z: number) => Math.max(ZOOM_MIN, Math.min(ZOOM_MAX, z));

  // ajuste inicial: a árvore cabe na largura da janela
  useLayoutEffect(() => {
    if (dados && !ajusteInicialFeito.current && largura > 0) {
      ajusteInicialFeito.current = true;
      setZoom(limitar(zoomParaLargura()));
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [dados, largura]);

  // Ctrl + roda do mouse = zoom
  useEffect(() => {
    const el = areaRef.current;
    if (!el) return;
    const aoRolar = (e: WheelEvent) => {
      if (!e.ctrlKey) return;
      e.preventDefault();
      setZoom((z) => limitar(z * (e.deltaY < 0 ? 1.1 : 1 / 1.1)));
    };
    el.addEventListener('wheel', aoRolar, { passive: false });
    return () => el.removeEventListener('wheel', aoRolar);
  }, [dados]);

  if (pessoas.isPending || unioes.isPending || filiacoes.isPending) return <p>Carregando…</p>;
  const erro = pessoas.error ?? unioes.error ?? filiacoes.error;
  if (erro) return <p className="erro">Erro: {erro.message}</p>;
  if (!dados) return null;

  const { layout, porId, curtos, semLigacao } = dados;
  const filtro = semAcento(busca.trim());
  const combina = (id: number) =>
    filtro === '' || semAcento(porId.get(id)?.nome ?? '').includes(filtro);

  const barra = (
    <div className="arvore-barra">
      <div className="arvore-barra-grupo">
        <button type="button" className="secundario" onClick={() => setZoom((z) => limitar(z / 1.15))}>
          −
        </button>
        <button type="button" className="secundario" onClick={() => setZoom(limitar(zoomParaTudo()))}>
          Ajustar
        </button>
        <button type="button" className="secundario" onClick={() => setZoom((z) => limitar(z * 1.15))}>
          +
        </button>
        <span className="arvore-zoom">{Math.round(zoom * 100)}%</span>
      </div>
      <label className="arvore-foco">
        Linha direta de
        <select
          value={foco ?? ''}
          onChange={(e) => setFocoEscolhido(e.target.value === '' ? null : Number(e.target.value))}
        >
          <option value="">(ninguém)</option>
          {[...layout.x.keys()].map((id) => (
            <option key={id} value={id}>
              {porId.get(id)?.nome ?? `#${id}`}
            </option>
          ))}
        </select>
      </label>
      <input
        className="arvore-busca"
        placeholder="⌕  Buscar na árvore..."
        value={busca}
        onChange={(e) => setBusca(e.target.value)}
      />
    </div>
  );

  if (layout.x.size === 0) {
    return (
      <section>
        <h2>Árvore</h2>
        <p>
          Ainda não há ligações para desenhar. Cadastre casais e vínculos de pais e filhos na tela{' '}
          <strong>Famílias</strong>.
        </p>
      </section>
    );
  }

  const nomeFoco = foco !== null ? (porId.get(foco)?.nome.split(' ')[0] ?? '') : '';

  return (
    <section>
      <h2>Árvore</h2>
      {barra}
      {layout.avisos.length > 0 && (

      <div className="arvore-area" ref={areaRef}>

          width={largura * zoom}
          height={altura * zoom}

          <defs>

              <stop offset="0%" stopColor="#eadcc0" />
              <stop offset="100%" stopColor="#f3e8d0" />
            </linearGradient>
            <filter id="sombra" x="-30%" y="-30%" width="160%" height="170%">
              <feDropShadow dx="0" dy="3" stdDeviation="3.5" floodColor="#503214" floodOpacity="0.28" />
            </filter>
            {[...layout.x.keys()].map((id) => {
              const cx = layout.x.get(id)! + MARGEM_ROTULOS;
              const cy = topoDaLinha(layout.linha.get(id)!) + RAIO_ANEL;
              return (
                <clipPath id={`foto-${id}`} key={id}>
                  <circle cx={cx} cy={cy} r={RAIO_FOTO} />
                </clipPath>
              );
            })}
          </defs>

          {/* fundo */}
          <rect width={largura} height={altura} fill="url(#fundo-radial)" />
          <image
            href="/fundo_arvore.png"
            width={largura}
            height={altura}
            preserveAspectRatio="none"
          />

          {/* ramos e casais */}
          {layout.casais.map((c, i) => {
            const cx = c.centroX + MARGEM_ROTULOS;
            const topo = topoDaLinha(c.linha);
            const yFoto = topo + RAIO_ANEL;
            let origemY: number;
            const partes: ReactNode[] = [];

            if (c.direitaId !== null) {
              const xe = layout.x.get(c.esquerdaId)! + MARGEM_ROTULOS;

              partes.push(
                <line
                  key="liga"
                  x1={Math.min(xe, xd)}
                  y1={yFoto}
                  x2={Math.max(xe, xd)}
                  y2={yFoto}

                  d={`M ${cx} ${origemY} C ${cx} ${origemY + m}, ${fx} ${fy - m}, ${fx} ${fy}`}

                  stroke={COR_RAMO}
                  strokeWidth={2.2}
                  strokeLinecap="round"
                />,
              );
            });

          })}

          {/* corações */}
          {layout.casais
            .filter((c) => c.direitaId !== null)

                  <text
                    x={cx}
                    y={cy + 4.5}
                    textAnchor="middle"
                    fontSize={12}
                    fill="#b3263e"
                    style={{ userSelect: 'none' }}
                  >
                    ♥
                  </text>
                </g>
              );
            })}

          {/* pessoas */}
          {[...layout.x.entries()].map(([id, x]) => {
            const p = porId.get(id);
            if (!p) return null;
            const cx = x + MARGEM_ROTULOS;
            const topo = topoDaLinha(layout.linha.get(id)!);
            const cy = topo + RAIO_ANEL;
            const destaque = linhaDireta.has(id);
            const nome = curtos.get(id) ?? p.nome;

              <g key={id} opacity={combina(id) ? 1 : 0.22}>
                <title>{p.nome}</title>

                  cx={cx}
                  cy={cy}
                  r={RAIO_ANEL}
                  fill={destaque ? '#b8892f' : '#fbf5e8'}
                  stroke={COR_RAMO}

                {p.foto ? (

                    x={cx - RAIO_FOTO}
                    y={cy - RAIO_FOTO}
                    width={2 * RAIO_FOTO}
                    height={2 * RAIO_FOTO}
                    preserveAspectRatio="xMidYMid slice"
                    clipPath={`url(#foto-${id})`}
                  />
                ) : (
                  <g clipPath={`url(#foto-${id})`}>
                    <circle cx={cx} cy={cy} r={RAIO_FOTO} fill="#cbd5e1" />
                    <circle cx={cx} cy={cy - 7} r={11} fill="#1e2a3a" />
                    <ellipse cx={cx} cy={cy + RAIO_FOTO - 2} rx={21} ry={20} fill="#1e2a3a" />
                  </g>
                )}
                <foreignObject
                  x={cx - LARGURA_CARTAO / 2}
                  y={topo + 2 * RAIO_ANEL + 5}
                  width={LARGURA_CARTAO}
                  height={64}
                >
                  <div className="arvore-nome-caixa">
                    <span className={destaque ? 'arvore-nome destaque' : 'arvore-nome'}>{nome}</span>
                  </div>
                </foreignObject>
              </g>
            );
          })}

          {/* rótulos das gerações */}
          {Array.from({ length: layout.totalLinhas }, (_, i) => {
            const titulo =
              layout.totalLinhas === 5 ? TITULOS_5[i]! : `${ROMANOS[i + 1] ?? i + 1} Geração`;
            const linhas = titulo.split('\n');
            const y0 = topoDaLinha(i) + 8;
            const h = 18 * linhas.length + 16;
            const r = Math.min(28, h / 2);
            return (
              <g key={`ger-${i}`} pointerEvents="none">
                <path
                  d={`M 0 ${y0} H ${140 - r} A ${r} ${r} 0 0 1 140 ${y0 + r} V ${y0 + h - r} A ${r} ${r} 0 0 1 ${140 - r} ${y0 + h} H 0 Z`}
                  fill="url(#rotulo-grad)"
                />
                {linhas.map((t, k) => (
                  <text
                    key={k}
                    x={75}
                    y={y0 + 16 + 4 + k * 18}
                    textAnchor="middle"
                    fontSize={14}
                    fill="#5a4128"
                    fontFamily="Georgia, serif"
                  >
                    {t}
                  </text>
                ))}
              </g>
            );
          })}

          {/* legenda */}
          <g transform={`translate(24 ${altura - 64})`} pointerEvents="none">
            <rect width={470} height={40} rx={20} fill="#fbf5e8" fillOpacity={0.92} stroke="#d8bd8c" />
            <circle cx={26} cy={20} r={9} fill="#fbf5e8" stroke="#b8892f" strokeWidth={3} />
            <text x={42} y={25} fontSize={12} fill="#5a4128" fontFamily="Georgia, serif">
              {nomeFoco ? `Linha direta de ${nomeFoco}` : 'Linha direta'}
            </text>
            <text x={190} y={26} fontSize={13} fill="#b3263e">
              ♥
            </text>
            <text x={208} y={25} fontSize={12} fill="#5a4128" fontFamily="Georgia, serif">
              Casal · os ramos descem para os filhos
            </text>
          </g>
        </svg>
      </div>

      {semLigacao.length > 0 && (
        <p className="arvore-sem-ligacao">
          <strong>Sem ligações (não aparecem na árvore):</strong>{' '}
          {semLigacao.map((p) => p.nome).join(' · ')}
        </p>
      )}
    </section>
  );
}

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';

type PessoaDto = {
  id: number;
  nome: string;
  dataNascimento: string | null;
  sexo: 'M' | 'F' | null;
};
type UniaoDto = { id: number; pessoaAId: number; pessoaBId: number };
type FiliacaoDto = { id: number; filhoId: number; genitorId: number; papel: 'PAI' | 'MAE' };

export function Familias() {

  const unioes = useQuery({ queryKey: ['unioes'], queryFn: () => api<UniaoDto[]>('/unioes') });
  const filiacoes = useQuery({
    queryKey: ['filiacoes'],
    queryFn: () => api<FiliacaoDto[]>('/filiacoes'),
  });

  const listaUnioes = unioes.data ?? [];
  const listaFiliacoes = filiacoes.data ?? [];

  let mapaNomesCurtos = new Map<number, string>();
  try {
    if (typeof nomesCurtos === 'function' && listaPessoas.length > 0) {
      mapaNomesCurtos = nomesCurtos(listaPessoas);
    }
  } catch {
    // Caso ocorra algum erro na função de domínio, usa o nome normal
  }

  const nome = (id: number) =>
    mapaNomesCurtos.get(id) ?? listaPessoas.find((p) => p.id === id)?.nome ?? `#${id}`;

  const pessoasEmCasal = useMemo(() => {
    const ids = new Set<number>();

      ids.add(u.pessoaBId);
    }
    return ids;
  }, [listaUnioes]);

  const genitoresPorFilho = useMemo(() => {
    const mapa = new Map<number, { paiId?: number; maeId?: number }>();
    for (const f of listaFiliacoes) {
      const atual = mapa.get(f.filhoId) ?? {};
      if (f.papel === 'PAI') atual.paiId = f.genitorId;
      if (f.papel === 'MAE') atual.maeId = f.genitorId;
      mapa.set(f.filhoId, atual);
    }
    return mapa;
  }, [listaFiliacoes]);

  const [pessoaB, setPessoaB] = useState('');

      (p) => p?.sexo === 'F',
    ) ?? mapaPessoas.get(uniaoAtual.pessoaBId)
    : soloAtual;

  const anoNascimentoMae = maeAtual?.dataNascimento ? Number(maeAtual.dataNascimento.substring(0, 4)) : null;

  // - Evita que o próprio genitor seja selecionado como filho
  // - Se for Casal: oculta filhos que já tenham AMBOS os pais cadastrados
  // - Se for Pai/Mãe Solo: oculta filhos que já tenham aquele genitor específico cadastrado
  // - Respeita a regra de idade mínima (10 anos de diferença)
  const filhosDisponiveis = listaPessoas.filter((p) => {
    if (uniaoAtual && (p.id === uniaoAtual.pessoaAId || p.id === uniaoAtual.pessoaBId)) return false;
    if (soloAtual && p.id === soloAtual.id) return false;

    const g = genitoresPorFilho.get(p.id);

    if (uniaoAtual) {
      // Se for casal, não mostra se já tem pai E mãe definidos
      if (g?.paiId && g?.maeId) return false;
    } else if (soloAtual) {
      // Se for solo, não mostra se já tem este genitor específico cadastrado
      if (papelSolo === 'PAI' && g?.paiId === soloAtual.id) return false;
      if (papelSolo === 'MAE' && g?.maeId === soloAtual.id) return false;
    }

    if (!p.dataNascimento) return true;
    const anoNascimentoFilho = Number(p.dataNascimento.substring(0, 4));
    return anoNascimentoFilho >= anoNascimentoMae + 10;

  const gruposFamiliares = useMemo(() => {

      const atual = genitoresTemp.get(f.filhoId) ?? {};
      if (f.papel === 'PAI') {
        atual.paiId = f.genitorId;
        atual.paiFiliacaoId = f.id;
      } else {
        atual.maeId = f.genitorId;
        atual.maeFiliacaoId = f.id;

      genitoresTemp.set(f.filhoId, atual);
    }

    for (const [filhoId, g] of genitoresTemp.entries()) {
      const chave = `${g.paiId ?? 'x'}-${g.maeId ?? 'x'}`;
      const filiacaoIds = [g.paiFiliacaoId, g.maeFiliacaoId].filter((id): id is number => id !== undefined);

      let grupo = mapa.get(chave);
      if (!grupo) {
        grupo = { chave, paiId: g.paiId, maeId: g.maeId, filhos: [] };
        mapa.set(chave, grupo);
      }
      grupo.filhos.push({ filhoId, filiacaoIds });
    }

    return Array.from(mapa.values());
  }, [listaFiliacoes]);

  return (

      <h2>Casais</h2>
      <p style={{ margin: '0 0 12px', fontSize: '0.9rem' }}>
        Pai ou mãe <strong>solo</strong> não precisa de casal: use o bloco “Pais e Filhos” mais abaixo.
      </p>
      <form className="form-pessoa" onSubmit={(e) => { e.preventDefault(); criarUniao.mutate(); }}>

        <select value={pessoaA} onChange={(e) => setPessoaA(e.target.value)} required>
          <option value="">Selecione o Pai</option>
          {candidatasPessoaA.map((p) => (
            <option key={p.id} value={p.id}>

import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';
import { useMemo, useState, type FormEvent } from 'react';
import { api } from '../api/cliente';

type PessoaDto = {
  id: number;
  nome: string;
  dataNascimento: string | null;
  sexo: 'M' | 'F' | null;
  foto: string | null;
};

async function enviarArquivo(id: number, arquivo: File) {
  const dados = new FormData();
  dados.append('foto', arquivo);
  const resp = await fetch(`/api/pessoas/${id}/foto`, { method: 'POST', body: dados });
  if (!resp.ok) {
    const corpo = (await resp.json().catch(() => null)) as { erro?: string } | null;
    throw new Error(corpo?.erro ?? `Erro ${resp.status}`);
  }
}

type Form = { nome: string; dataNascimento: string; sexo: '' | 'M' | 'F' };
const vazio: Form = { nome: '', dataNascimento: '', sexo: '' };

export function Pessoas() {
  const qc = useQueryClient();
  const [form, setForm] = useState<Form>(vazio);
  const [editandoId, setEditandoId] = useState<number | null>(null);
  const [fotoEscolhida, setFotoEscolhida] = useState<File | null>(null);
  const [chaveInputFoto, setChaveInputFoto] = useState(0);
  const previa = useMemo(
    () => (fotoEscolhida ? URL.createObjectURL(fotoEscolhida) : null),
    [fotoEscolhida],
  );

  const lista = useQuery({ queryKey: ['pessoas'], queryFn: () => api<PessoaDto[]>('/pessoas') });

  const limpar = () => {
    setForm(vazio);
    setEditandoId(null);
    setFotoEscolhida(null);
    setChaveInputFoto((k) => k + 1);
  };

  const salvar = useMutation({
    mutationFn: async () => {
      const corpo = JSON.stringify({
        nome: form.nome,
        dataNascimento: form.dataNascimento || null,
        sexo: form.sexo || null,
      });
      const salva =
        editandoId === null
          ? await api<PessoaDto>('/pessoas', { method: 'POST', body: corpo })
          : await api<PessoaDto>(`/pessoas/${editandoId}`, { method: 'PUT', body: corpo });
      if (fotoEscolhida) await enviarArquivo(salva.id, fotoEscolhida);
      return salva;
    },
    onSuccess: () => {
      limpar();
      void qc.invalidateQueries({ queryKey: ['pessoas'] });
    },
  });

  const excluir = useMutation({
    mutationFn: (id: number) => api<void>(`/pessoas/${id}`, { method: 'DELETE' }),
    onSuccess: () => void qc.invalidateQueries({ queryKey: ['pessoas'] }),
  });

  const enviarFoto = useMutation({
    mutationFn: ({ id, arquivo }: { id: number; arquivo: File }) => enviarArquivo(id, arquivo),
    onSuccess: () => void qc.invalidateQueries({ queryKey: ['pessoas'] }),
  });

  const removerFoto = useMutation({
    mutationFn: (id: number) => api<PessoaDto>(`/pessoas/${id}/foto`, { method: 'DELETE' }),
    onSuccess: () => void qc.invalidateQueries({ queryKey: ['pessoas'] }),
  });

  const aoEnviar = (e: FormEvent) => {
    e.preventDefault();
    salvar.mutate();
  };

  const editar = (p: PessoaDto) => {
    setEditandoId(p.id);
    setForm({ nome: p.nome, dataNascimento: p.dataNascimento ?? '', sexo: p.sexo ?? '' });
  };

  return (
    <section>
      <h2>Pessoas</h2>

      <form className="form-pessoa" onSubmit={aoEnviar}>
        <input
          placeholder="Nome completo"
          value={form.nome}
          onChange={(e) => setForm({ ...form, nome: e.target.value })}
          required
          maxLength={150}
        />
        <input
          type="date"
          value={form.dataNascimento}
          onChange={(e) => setForm({ ...form, dataNascimento: e.target.value })}
        />
        <select
          value={form.sexo}
          onChange={(e) => setForm({ ...form, sexo: e.target.value as Form['sexo'] })}
        >
          <option value="">Sexo (opcional)</option>
          <option value="M">Masculino</option>
          <option value="F">Feminino</option>
        </select>
        <label className="botao secundario">
          {fotoEscolhida ? 'Trocar foto' : 'Escolher foto'}
          <input
            key={chaveInputFoto}
            type="file"
            accept="image/jpeg,image/png,image/webp"
            hidden
            onChange={(e) => setFotoEscolhida(e.target.files?.[0] ?? null)}
          />
        </label>
        {previa && <img className="foto-mini" src={previa} alt="Prévia da foto" />}
        <button type="submit" disabled={salvar.isPending}>
          {editandoId === null ? 'Adicionar' : 'Salvar alterações'}
        </button>
        {editandoId !== null && (
          <button type="button" className="secundario" onClick={limpar}>
            Cancelar
          </button>
        )}
      </form>
      {salvar.error && <p className="erro">{salvar.error.message}</p>}
      {excluir.error && <p className="erro">{excluir.error.message}</p>}
      {enviarFoto.error && <p className="erro">{enviarFoto.error.message}</p>}
      {removerFoto.error && <p className="erro">{removerFoto.error.message}</p>}

      {lista.isPending && <p>Carregando…</p>}
      {lista.error && <p className="erro">Erro: {lista.error.message}</p>}
      {lista.data && (
        <table className="tabela">
          <thead>
            <tr>
              <th>Foto</th>
              <th>Nome</th>
              <th>Nascimento</th>
              <th>Sexo</th>
              <th />
            </tr>
          </thead>
          <tbody>
            {lista.data.map((p) => (
              <tr key={p.id}>
                <td>
                  {p.foto ? (
                    <img className="foto-mini" src={`/api/uploads/${p.foto}`} alt={p.nome} />
                  ) : (
                    <span className="foto-mini vazia">{p.nome.charAt(0)}</span>
                  )}
                </td>
                <td>{p.nome}</td>
                <td>{p.dataNascimento ? p.dataNascimento.split('-').reverse().join('/') : '—'}</td>
                <td>{p.sexo ?? '—'}</td>
                <td>
                  <div className="acoes">
                  <label className="botao secundario">
                    {p.foto ? 'Trocar foto' : 'Enviar foto'}

                      type="file"
                      accept="image/jpeg,image/png,image/webp"

                        const arquivo = e.target.files?.[0];

                      }}

                  </label>
                  {p.foto && (
                    <button type="button" className="secundario" onClick={() => removerFoto.mutate(p.id)}>
                      Tirar foto

                  <button

      )}
    </section>
  );
}

# Arquitetura

```
raizes-memorias/
├─ packages/domain   TypeScript puro: tipos, nomes curtos, (depois) layout da árvore e parentesco
├─ apps/api          Fastify + Drizzle + PostgreSQL
│   └─ modules/<x>   routes → service → repository
└─ apps/web          React + Vite + TanStack Query
```

Regras:
- `domain` não importa nada de api/web nem de bibliotecas de runtime; testável sozinho.
- API em camadas: **routes** (HTTP + zod) → **service** (regras) → **repository** (SQL via Drizzle).
- Tudo por **id**, nunca por nome. Parentesco é **derivado**, não gravado.
- `createApp({ db })` permite testar a API com banco de teste.
- Web fala com a API só via `src/api/cliente.ts`; dados do servidor via TanStack Query.
- Fotos: caminho relativo em `pessoa.foto`, arquivos em `apps/api/uploads`.

# Comece aqui

## 1. O que baixar e instalar (uma vez só)

| Item | Para quê | Onde |
|---|---|---|
| **Node.js 22 LTS** | roda TypeScript, API e Vite | nodejs.org |
| **pnpm** | gerenciador de pacotes do monorepo | depois do Node: `corepack enable` (ou `npm i -g pnpm`) |
| **Git** | versionamento | git-scm.com |
| **PostgreSQL** | você já tem; crie o banco `raizes_v2` | pgAdmin/DBeaver para criar |
| VS Code + extensões **ESLint**, **Prettier**, (opcional) **Error Lens** | editor | |
| Docker Desktop (opcional) | alternativa ao Postgres local, via `docker-compose.yml` | |

Dica: coloque o projeto em pasta **sem acento, sem "&" e fora do OneDrive**, ex.: `C:\dev\raizes-memorias`.

## 2. Primeira execução

```powershell
cd C:\dev\raizes-memorias
pnpm install
copy apps\api\.env.example apps\api\.env     # edite usuário/senha do Postgres
pnpm db:generate                              # gera a migração a partir do schema

pnpm typecheck
pnpm test
pnpm dev                                      # API :3333 + Web :5173
```

Abra http://localhost:5173 — a tela Início mostra se o banco conectou.

## 3. Se algo falhar
As versões das dependências foram escritas sem poder rodar `pnpm install` aqui. Se algum pacote reclamar de versão, me mande a mensagem de erro que eu ajusto.

## 4. Segurança
Nunca coloque senha no código nem no git. Use só o `.env` (já ignorado). Como a senha antiga está no `ConexaoBanco.java`, considere trocá-la.

# Modelo de dados

- **pessoa**(id, nome, data_nascimento, sexo M/F opcional, foto, criado_em)
- **uniao**(id, pessoa_a_id, pessoa_b_id) — CHECK a<b, UNIQUE(a,b): um casal tem uma linha só
- **filiacao**(id, filho_id, genitor_id, papel PAI|MAE) — UNIQUE(filho,papel), CHECK filho<>genitor

Tudo com ON DELETE CASCADE. Irmão, avô, tio, primo etc. são calculados no código.

import type { Filiacao } from '../tipos';

/**
 * Verdadeiro se `ancestralId` é ancestral (pai, avô, bisavô...) de `pessoaId`.
 * Usado para impedir ciclos: ninguém pode ser pai de um de seus próprios ancestrais.
 */
export function ehAncestral(
  filiacoes: Pick<Filiacao, 'filhoId' | 'genitorId'>[],
  ancestralId: number,
  pessoaId: number,
): boolean {
  const genitoresDe = new Map<number, number[]>();
  for (const f of filiacoes) {
    const lista = genitoresDe.get(f.filhoId) ?? [];
    lista.push(f.genitorId);
    genitoresDe.set(f.filhoId, lista);
  }
  const visitados = new Set<number>();
  const pilha = [pessoaId];
  while (pilha.length > 0) {
    const atual = pilha.pop()!;
    for (const g of genitoresDe.get(atual) ?? []) {
      if (g === ancestralId) return true;
      if (!visitados.has(g)) {
        visitados.add(g);
        pilha.push(g);
      }
    }
  }
  return false;
}

import type { Familia, Filiacao, Uniao } from '../tipos';

type UniaoMin = Pick<Uniao, 'pessoaAId' | 'pessoaBId'>;
type FiliacaoMin = Pick<Filiacao, 'filhoId' | 'genitorId' | 'papel'>;

/**
 * Transforma as ligações do banco (uniões + filiações, tudo por id) nas
 * "famílias" que o layout da árvore entende: pai, mãe e filhos.
 *
 * Regras:
 * - Todo casal cadastrado vira uma família, mesmo sem filhos.
 * - Os filhos entram no casal formado pelos seus dois genitores.
 * - Se um filho só tem UM genitor cadastrado e esse genitor tem exatamente
 *   UMA união, o parceiro é considerado o outro genitor (só para desenhar).
 * - Pessoas sem nenhuma ligação não entram (a tela as lista à parte).
 */
export function montarFamilias(unioes: UniaoMin[], filiacoes: FiliacaoMin[]): Familia[] {
  // parceiros de cada pessoa
  const parceiros = new Map<number, number[]>();
  const addParceiro = (a: number, b: number) => {
    const l = parceiros.get(a) ?? [];
    if (!l.includes(b)) l.push(b);
    parceiros.set(a, l);
  };
  for (const u of unioes) {
    addParceiro(u.pessoaAId, u.pessoaBId);
    addParceiro(u.pessoaBId, u.pessoaAId);
  }

  // pai e mãe de cada filho (na ordem em que os filhos aparecem)
  const genitores = new Map<number, { pai: number | null; mae: number | null }>();
  for (const f of filiacoes) {
    const g = genitores.get(f.filhoId) ?? { pai: null, mae: null };
    if (f.papel === 'PAI') g.pai = f.genitorId;
    else g.mae = f.genitorId;
    genitores.set(f.filhoId, g);
  }

  // completa o genitor que falta quando só há um parceiro possível
  for (const g of genitores.values()) {
    if (g.pai !== null && g.mae === null) {
      const ps = parceiros.get(g.pai);
      if (ps && ps.length === 1) g.mae = ps[0]!;
    } else if (g.mae !== null && g.pai === null) {
      const ps = parceiros.get(g.mae);
      if (ps && ps.length === 1) g.pai = ps[0]!;
    }
  }

  const familias = new Map<string, Familia>();
  const chaveCasal = (a: number, b: number) => `${Math.min(a, b)}-${Math.max(a, b)}`;

  // 1) casais cadastrados (na ordem de cadastro)
  for (const u of unioes) {
    familias.set(chaveCasal(u.pessoaAId, u.pessoaBId), {
      paiId: u.pessoaAId,
      maeId: u.pessoaBId,
      filhosIds: [],
    });
  }

  // 2) filhos
  for (const [filho, g] of genitores) {
    let chave: string;
    if (g.pai !== null && g.mae !== null) chave = chaveCasal(g.pai, g.mae);
    else chave = `solo-${g.pai ?? g.mae}`;

    let fam = familias.get(chave);
    if (!fam) {
      fam = { paiId: g.pai, maeId: g.mae, filhosIds: [] };
      familias.set(chave, fam);
    } else {
      // garante pai/mãe corretos (o casal pode ter vindo só da união)
      if (g.pai !== null) fam.paiId = g.pai;
      if (g.mae !== null) fam.maeId = g.mae;
    }
    if (!fam.filhosIds.includes(filho)) fam.filhosIds.push(filho);
  }

  return [...familias.values()];
}

import type { Familia } from '../tipos';

/**
 * Calcula a posição de cada pessoa numa árvore genealógica desenhada em
 * linhas (gerações). Não depende de tela nem de banco: recebe famílias
 * (pai, mãe, filhos) e devolve coordenadas.
 *
 * Ideia: cada casal vira uma "unidade". As unidades são distribuídas em
 * linhas (geração) e depois ajustadas várias vezes para que o casal fique
 * centralizado sobre os filhos e os filhos sob os pais.
 */

/** Largura reservada para cada pessoa (foto + nome). */
export const LARGURA_NO = 100;
/** Distância entre os dois cônjuges (centro a centro). */
export const DISTANCIA_CASAL = 210;
/** Altura de cada linha de geração. */
export const ALTURA_LINHA = 200;
export const MARGEM = 60;

/** Casal (ou pessoa sozinha) já posicionado, com os filhos que saem dele. */
export interface CasalPosicionado {
  esquerdaId: number;
  /** null se for uma pessoa sozinha */
  direitaId: number | null;
  filhosIds: number[];
  linha: number;
  /** posição do coração (ou da pessoa sozinha) */
  centroX: number;
}

export interface ResultadoLayout {
  x: Map<number, number>;
  linha: Map<number, number>;
  /** ids dos pais de cada pessoa que tem pais na árvore */
  pais: Map<number, number[]>;
  casais: CasalPosicionado[];
  totalLinhas: number;
  largura: number;
  /** problemas encontrados nos dados (ex.: pessoa em dois casais) */
  avisos: string[];
}

class Unidade {
  membros: number[] = [];
  filhos: number[] = [];
  linha = 0;
  cx = 0;
  emitida = false;
  /** true = o segundo membro fica à esquerda */
  trocada = false;

  get casal(): boolean {
    return this.membros.length === 2;
  }
  get esquerda(): number {
    return this.casal && this.trocada ? this.membros[1]! : this.membros[0]!;
  }
  get direita(): number | null {
    return this.casal ? (this.trocada ? this.membros[0]! : this.membros[1]!) : null;
  }
  get largura(): number {
    return this.casal ? DISTANCIA_CASAL + LARGURA_NO : LARGURA_NO;
  }
  xDe(pessoa: number): number {
    if (!this.casal) return this.cx;
    return pessoa === this.esquerda ? this.cx - DISTANCIA_CASAL / 2 : this.cx + DISTANCIA_CASAL / 2;
  }
}

export function calcularLayout(familias: Familia[]): ResultadoLayout {
  const avisos: string[] = [];
  const unidadeDe = new Map<number, Unidade>();
  const unidades: Unidade[] = [];

  // 1) casais (uma unidade por casal) ------------------------------------
  for (const f of familias) {
    const pais: number[] = [];
    if (f.paiId !== null) pais.push(f.paiId);
    if (f.maeId !== null && f.maeId !== f.paiId) pais.push(f.maeId);
    if (pais.length === 0) continue;

    let existente: Unidade | null = null;
    let conflito = false;
    for (const p of pais) {
      const u = unidadeDe.get(p);
      if (u) {
        if (existente === null) existente = u;
        else if (existente !== u) conflito = true;
      }
    }

    if (existente) {
      const mesmoCasal =
        !conflito &&
        pais.every((p) => existente!.membros.includes(p)) &&
        existente.membros.every((m) => pais.includes(m));
      if (mesmoCasal) {
        for (const c of f.filhosIds) if (!existente.filhos.includes(c)) existente.filhos.push(c);
      } else {
        avisos.push(`Família ignorada: uma das pessoas já aparece em outro casal (${pais.join(' e ')})`);
      }
      continue;
    }

    const u = new Unidade();
    u.membros.push(...pais);
    for (const c of f.filhosIds) if (!u.filhos.includes(c)) u.filhos.push(c);
    unidades.push(u);
    for (const p of pais) unidadeDe.set(p, u);
  }

  // 2) pessoas soltas (só aparecem como filhos) --------------------------
  const todas = new Set<number>();
  for (const f of familias) {
    if (f.paiId !== null) todas.add(f.paiId);
    if (f.maeId !== null) todas.add(f.maeId);
    for (const c of f.filhosIds) todas.add(c);
  }
  for (const p of todas) {
    if (!unidadeDe.has(p)) {
      const u = new Unidade();
      u.membros.push(p);
      unidades.push(u);
      unidadeDe.set(p, u);
    }
  }

  // 3) de quais pais cada pessoa é filha ---------------------------------
  const paisDe = new Map<number, Unidade>();
  for (const u of unidades) {
    for (const c of u.filhos) {
      if (unidadeDe.get(c) === u || paisDe.has(c)) continue;
      paisDe.set(c, u);
    }
  }
  for (const u of unidades) u.filhos = u.filhos.filter((c) => paisDe.get(c) === u);

  // 4) linhas (gerações) -------------------------------------------------
  const limite = unidades.length * 4 + 10;
  let mudou: boolean;
  let voltas = 0;
  do {
    mudou = false;
    for (const u of unidades) {
      if (u.filhos.length === 0) continue;
      // todos os filhos de um casal ficam na mesma linha, logo abaixo dos pais
      let alvo = u.linha + 1;
      for (const c of u.filhos) alvo = Math.max(alvo, unidadeDe.get(c)!.linha);
      for (const c of u.filhos) {
        const cu = unidadeDe.get(c)!;
        if (cu.linha !== alvo) {
          cu.linha = alvo;
          mudou = true;
        }
      }
    }
  } while (mudou && voltas++ < limite);

  // quem não tem pais desce até ficar logo acima dos filhos
  voltas = 0;
  do {
    mudou = false;
    for (const u of unidades) {
      if (!ehRaiz(u, paisDe) || u.filhos.length === 0) continue;
      let min = Number.MAX_SAFE_INTEGER;
      for (const c of u.filhos) min = Math.min(min, unidadeDe.get(c)!.linha);
      if (min - 1 > u.linha) {
        u.linha = min - 1;
        mudou = true;
      }
    }
  } while (mudou && voltas++ < limite);

  let totalLinhas = 0;
  for (const u of unidades) totalLinhas = Math.max(totalLinhas, u.linha + 1);

  // 5) ordem dentro de cada linha (busca em profundidade) ----------------
  const linhas: Unidade[][] = Array.from({ length: totalLinhas }, () => []);

  const raizes = unidades.filter((u) => ehRaiz(u, paisDe));
  raizes.sort((a, b) => a.linha - b.linha); // estável: mantém a ordem de cadastro

  for (const r of raizes) visitar(r, linhas, unidadeDe, paisDe);
  for (const u of unidades) visitar(u, linhas, unidadeDe, paisDe);

  // posição inicial: lado a lado
  for (const linha of linhas) {
    let x = 0;
    let anterior: Unidade | null = null;
    for (const u of linha) {
      if (anterior) x += (anterior.largura + u.largura) / 2;
      u.cx = x;
      anterior = u;
    }
  }

  // 6) ajuste: centraliza casais sobre filhos e filhos sob pais ----------
  const indice = new Map<Unidade, number>();
  unidades.forEach((u, i) => indice.set(u, i));

  for (let volta = 0; volta < 120; volta++) {
    for (const u of unidades) orientar(u, paisDe);

    const desejado = unidades.map((u) => centroDesejado(u, unidadeDe, paisDe));

    for (const linha of linhas) {
      if (linha.length === 0) continue;
      const alvo = linha.map((u) => desejado[indice.get(u)!]!);
      const posicoes = distribuir(linha, alvo);
      linha.forEach((u, i) => {
        u.cx += 0.5 * (posicoes[i]! - u.cx);
      });
    }
  }
  // última passada sem suavização: garante o espaçamento mínimo
  for (const linha of linhas) {
    if (linha.length === 0) continue;
    const p = distribuir(
      linha,
      linha.map((u) => u.cx),
    );
    linha.forEach((u, i) => {
      u.cx = p[i]!;
    });
  }

  // 7) normaliza (margem esquerda) e monta o resultado -------------------
  let minEsq = Number.MAX_VALUE;
  let maxDir = -Number.MAX_VALUE;
  for (const u of unidades) {
    minEsq = Math.min(minEsq, u.cx - u.largura / 2);
    maxDir = Math.max(maxDir, u.cx + u.largura / 2);
  }
  const deslocamento = MARGEM - minEsq;

  const r: ResultadoLayout = {
    x: new Map(),
    linha: new Map(),
    pais: new Map(),
    casais: [],
    totalLinhas,
    largura: maxDir - minEsq + 2 * MARGEM,
    avisos,
  };

  for (const u of unidades) {
    u.cx += deslocamento;
    for (const m of u.membros) {
      r.x.set(m, u.xDe(m));
      r.linha.set(m, u.linha);
    }
    if (u.casal || u.filhos.length > 0) {
      r.casais.push({
        esquerdaId: u.esquerda,
        direitaId: u.direita,
        filhosIds: [...u.filhos],
        linha: u.linha,
        centroX: u.cx,
      });
    }
  }
  for (const [filho, u] of paisDe) r.pais.set(filho, [...u.membros]);
  return r;
}

function ehRaiz(u: Unidade, paisDe: Map<number, Unidade>): boolean {
  return !u.membros.some((m) => paisDe.has(m));
}

function visitar(
  u: Unidade,
  linhas: Unidade[][],
  unidadeDe: Map<number, Unidade>,
  paisDe: Map<number, Unidade>,
): void {
  if (u.emitida) return;
  u.emitida = true;
  linhas[u.linha]!.push(u);

  // filhos já colocados primeiro, depois os comuns, e por último os que
  // ainda têm outra família (cônjuge) a ser desenhada mais à direita
  const prioridade = (c: number): number => {
    const cu = unidadeDe.get(c)!;
    if (cu.emitida) return 0;
    for (const m of cu.membros) {
      const outro = paisDe.get(m);
      if (outro && outro !== u && !outro.emitida) return 2;
    }
    return 1;
  };
  const chaves = new Map(u.filhos.map((c) => [c, prioridade(c)]));
  const filhos = [...u.filhos].sort((a, b) => chaves.get(a)! - chaves.get(b)!);
  for (const c of filhos) visitar(unidadeDe.get(c)!, linhas, unidadeDe, paisDe);
}

/** Escolhe qual cônjuge fica à esquerda: o que tem os pais mais à esquerda. */
function orientar(u: Unidade, paisDe: Map<number, Unidade>): void {
  if (!u.casal) return;
  const pa = paisDe.get(u.membros[0]!);
  const pb = paisDe.get(u.membros[1]!);
  let trocar: boolean | null = null;
  if (pa && pb) {
    if (Math.abs(pa.cx - pb.cx) > 20) trocar = pb.cx < pa.cx;
  } else if (pa) {
    if (Math.abs(pa.cx - u.cx) > 20) trocar = pa.cx > u.cx;
  } else if (pb) {
    if (Math.abs(pb.cx - u.cx) > 20) trocar = pb.cx < u.cx;
  }
  if (trocar !== null) u.trocada = trocar;
}

function centroDesejado(
  u: Unidade,
  unidadeDe: Map<number, Unidade>,
  paisDe: Map<number, Unidade>,
): number {
  let soma = 0;
  let n = 0;

  for (const m of u.membros) {
    const pu = paisDe.get(m);
    if (pu) {
      soma += pu.cx - (u.xDe(m) - u.cx);
      n++;
    }
  }
  if (u.filhos.length > 0) {
    let min = Number.MAX_VALUE;
    let max = -Number.MAX_VALUE;
    for (const c of u.filhos) {
      const xc = unidadeDe.get(c)!.xDe(c);
      min = Math.min(min, xc);
      max = Math.max(max, xc);
    }
    soma += (min + max) / 2;
    n++;
  }
  return n === 0 ? u.cx : soma / n;
}

/**
 * Mantém a ordem da linha, respeita a distância mínima entre unidades
 * vizinhas e chega o mais perto possível das posições desejadas
 * (regressão isotônica / "pool adjacent violators").
 */
function distribuir(linha: Unidade[], alvo: number[]): number[] {
  const n = linha.length;
  const deslocamento = new Array<number>(n).fill(0);
  for (let i = 1; i < n; i++) {
    deslocamento[i] = deslocamento[i - 1]! + (linha[i - 1]!.largura + linha[i]!.largura) / 2;
  }

  const soma = new Array<number>(n).fill(0);
  const qtd = new Array<number>(n).fill(0);
  let blocos = 0;
  for (let i = 0; i < n; i++) {
    soma[blocos] = alvo[i]! - deslocamento[i]!;
    qtd[blocos] = 1;
    blocos++;
    while (blocos > 1 && soma[blocos - 2]! / qtd[blocos - 2]! > soma[blocos - 1]! / qtd[blocos - 1]!) {
      soma[blocos - 2]! += soma[blocos - 1]!;
      qtd[blocos - 2]! += qtd[blocos - 1]!;
      blocos--;
    }
  }

  const resultado = new Array<number>(n).fill(0);
  let i = 0;
  for (let b = 0; b < blocos; b++) {
    const media = soma[b]! / qtd[b]!;
    for (let k = 0; k < qtd[b]!; k++, i++) resultado[i] = media + deslocamento[i]!;
  }
  return resultado;
}

const CONECTORES = new Set(['de', 'da', 'do', 'das', 'dos', 'e']);

function normalizar(texto: string): string {
  return texto.normalize('NFD').replace(/\p{M}/gu, '').toLowerCase();
}

/** As primeiras n palavras "de verdade" (de/da/do não contam, mas ficam no texto). */
function comPalavras(completo: string, n: number): string {
  const partes = completo.trim().split(/\s+/);
  const usadas: string[] = [];
  let contadas = 0;

  for (const parte of partes) {
    const conector = CONECTORES.has(parte.toLowerCase());
    if (!conector && contadas === n) break;
    usadas.push(parte);
    if (!conector) contadas++;
  }
  while (usadas.length > 0 && CONECTORES.has(usadas[usadas.length - 1]!.toLowerCase())) {
    usadas.pop();
  }

  const texto = usadas.join(' ');
  // nomes muito curtos (ex.: "Zé") ganham a segunda palavra
  if (n === 1 && partes.length > 1 && texto.length <= 2) return comPalavras(completo, 2);
  return texto;
}

/**
 * Cada pessoa mostra o menor nome que a diferencia das outras
 * (Maria, Maria Celina, Maria do Socorro...). Portado do ArvoreController em Java.
 */
export function nomesCurtos(pessoas: { id: number; nome: string }[]): Map<number, string> {
  const resultado = new Map<number, string>();
  const validas = pessoas.filter((p) => p.nome.trim() !== '');

  for (let palavras = 1; palavras <= 12; palavras++) {
    const candidatos = new Map<number, string>();
    const contagem = new Map<string, number>();

    for (const p of validas) {
      const nome = comPalavras(p.nome, palavras);
      candidatos.set(p.id, nome);
      const chave = normalizar(nome);
      contagem.set(chave, (contagem.get(chave) ?? 0) + 1);
    }
    for (const p of validas) {
      if (resultado.has(p.id)) continue;
      const nome = candidatos.get(p.id)!;
      if (contagem.get(normalizar(nome)) === 1 || palavras === 12) resultado.set(p.id, nome);
    }
  }
  return resultado;
}
